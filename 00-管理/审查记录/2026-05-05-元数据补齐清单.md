## 扫描命令

```powershell
$root = (Get-Location).Path
$issues = @()
Get-ChildItem -Recurse -File -Filter *.md |
  Where-Object { $_.FullName -notmatch '\\.git\\|\\.obsidian\\|\\.codex-temp\\' } |
  ForEach-Object {
    $text = Get-Content -Raw -Encoding UTF8 -LiteralPath $_.FullName
    $rel = $_.FullName.Substring($root.Length + 1).Replace('\','/')
    $fm = [regex]::Match($text, '(?s)^---\r?\n(.*?)\r?\n---')
    if (-not $fm.Success) {
      $issues += [pscustomobject]@{file=$rel; issue='no-frontmatter'}
    } else {
      $y = $fm.Groups[1].Value
      if ($y -notmatch '(?m)^type:') { $issues += [pscustomobject]@{file=$rel; issue='missing-type'} }
      if ($y -notmatch '(?m)^domain:') { $issues += [pscustomobject]@{file=$rel; issue='missing-domain'} }
      if ($y -notmatch '(?m)^status:') { $issues += [pscustomobject]@{file=$rel; issue='missing-status'} }
      if ($y -notmatch '(?m)^tags:') { $issues += [pscustomobject]@{file=$rel; issue='missing-tags'} }
    }
  }
$issues | Group-Object issue | Sort-Object Name | Select-Object Name,Count
```
