<%*
// ========================================
// 自动标签模板 - 根据文件夹和内容智能添加标签
// ========================================

// 获取当前文件路径
const filePath = tp.file.path;
const folder = tp.file.folder(true);
const fileName = tp.file.title;

// 定义标签映射规则
let domainTags = [];
let typeTags = [];
let statusTags = [];
let priorityTags = [];

// 1. 根据文件夹自动添加领域标签
if (folder.includes("技术笔记") || folder.includes("技术沉淀")) {
    domainTags = ["#领域/技术"];
} else if (folder.includes("AI 学习") || folder.includes("AI")) {
    domainTags = ["#领域/AI"];
} else if (folder.includes("工作") || folder.includes("项目") || folder.includes("TODO")) {
    domainTags = ["#领域/工作"];
} else if (folder.includes("知识库")) {
    domainTags = ["#领域/知识"];
    priorityTags = ["#重要"];
} else if (folder.includes("思维笔记")) {
    domainTags = ["#领域/思维"];
} else if (folder.includes("数据")) {
    domainTags = ["#领域/技术"];
} else if (folder.includes("Clippings")) {
    domainTags = ["#领域/AI"];
    typeTags = ["#类型/文章"];
} else if (folder.includes("模版") || folder.includes("Templates")) {
    domainTags = [];
    typeTags = ["#类型/模板"];
} else if (folder.includes("周工作总结")) {
    domainTags = ["#领域/工作"];
    typeTags = ["#类型/汇总"];
} else if (folder.includes("Excalidraw")) {
    typeTags = ["#类型/卡片"];
} else {
    // 默认标记为待整理
    priorityTags = ["#待整理"];
}

// 2. 根据文件名判断类型
if (fileName.includes("模板") || fileName.includes("template")) {
    typeTags = ["#类型/模板"];
} else if (fileName.includes("日记") || fileName.includes("日记")) {
    typeTags = ["#类型/汇总"];
    domainTags = ["#领域/生活"];
} else if (fileName.includes("会议")) {
    typeTags = ["#类型/汇总"];
} else if (fileName.includes("项目") || fileName.includes("规划")) {
    typeTags = ["#类型/项目"];
    priorityTags = ["#重要"];
} else if (fileName.includes("笔记")) {
    typeTags = ["#类型/笔记"];
} else if (fileName.includes("命令") || fileName.includes("速查") || fileName.includes("指南")) {
    typeTags = ["#类型/资源"];
    priorityTags = ["#重要"];
} else if (fileName.match(/^\d{4}-\d{2}-\d{2}/)) {
    // 日期格式的文件名（日报）
    typeTags = ["#类型/汇总"];
    if (!domainTags.length) domainTags = ["#领域/生活"];
}

// 3. 如果还没有类型标签，默认为笔记
if (!typeTags.length && !domainTags.includes("#领域/技术")) {
    typeTags = ["#类型/笔记"];
}

// 4. 弹窗选择状态（可选）
const status = await tp.system.prompt(
    "选择笔记状态",
    ["进行中", "待处理", "已完成", "草稿", "跳过"],
    "进行中"
);

if (status !== "跳过") {
    statusTags = [`#状态/${status}`];
}

// 5. 合并所有标签
const allTags = [...domainTags, ...typeTags, ...statusTags, ...priorityTags];

// 6. 生成 frontmatter
const created = tp.date.now("YYYY-MM-DD HH:mm");
const modified = created;

tR += `---
created: ${created}
modified: ${modified}
tags:
`;

// 写入标签
if (allTags.length > 0) {
    allTags.forEach(tag => {
        tR += `  - ${tag.replace("#", "")}\n`;
    });
} else {
    tR += `  - 待整理\n`;
}

tR += `aliases: []
---

`;
%>
