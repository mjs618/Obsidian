<%*
// ========================================
// 快速添加标签宏
// 使用方法：在笔记中运行此模板
// ========================================

const editor = app.workspace.activeEditor;
if (!editor) {
    new Notice("请先打开一篇笔记");
    return;
}

// 常用标签选项
const tagOptions = [
    { label: "技术笔记", value: "#领域/技术 #类型/笔记" },
    { label: "AI 文章", value: "#领域/AI #类型/文章" },
    { label: "工作项目", value: "#领域/工作 #类型/项目 #重要" },
    { label: "工作总结", value: "#领域/工作 #类型/汇总" },
    { label: "知识库", value: "#领域/知识 #类型/笔记 #重要" },
    { label: "学习笔记", value: "#领域/思维 #类型/笔记" },
    { label: "灵感卡片", value: "#领域/思维 #类型/卡片 #灵感" },
    { label: "日常记录", value: "#领域/生活 #类型/汇总" },
    { label: "资源参考", value: "#领域/技术 #类型/资源 #重要" },
    { label: "待复习", value: "#待复习" },
    { label: "待整理", value: "#待整理" },
];

// 创建选择器
const selectedLabel = await tp.system.prompt(
    "选择要添加的标签组合",
    tagOptions.map(opt => opt.label),
    tagOptions[0].label
);

// 找到对应的标签
const selected = tagOptions.find(opt => opt.label === selectedLabel);
if (selected) {
    // 在光标位置插入标签
    editor.replaceSelection(` ${selected.value} `);
    new Notice(`已添加标签：${selected.value}`);
} else {
    new Notice("未选择标签");
}
%>
