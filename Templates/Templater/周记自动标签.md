const now = moment();

// 获取当前周数

const sunday = now.clone().endOf('isoWeek').format('YYYY-MM-DD');

    "领域/生活",

    `时间/${year}`,
    `时间/${now.format("YYYY-MM")}`,
    `时间/周记`,
    `时间/W${weekNum}`
];

// 生成 frontmatter
tR += `---

`;

    tR += `  - ${tag}\n`;
});

tR += `aliases: ["${year}年第${weekNum}周总结"]
week: ${weekNum}
year: ${year}
period: ${monday} ~ ${sunday}
---

`;
